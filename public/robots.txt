# studiounfolding.cc — crawl policy
# Default: allow everything. The /tracker app (job tracker + its API) and the
# /jobs board are private tools behind Cloudflare Access, not content, so keep
# them out of search and AI indexes.
#
# NOTE: the Disallow paths below deliberately have NO trailing slash. robots.txt
# matches on prefix, so "Disallow: /jobs" covers /jobs, /jobs/, /jobs?x=1 and
# every path underneath, while "Disallow: /jobs/" would have left the
# slash-less /jobs — which the site serves with a 200 — fully crawlable.

User-agent: *
Allow: /
Disallow: /tracker
Disallow: /jobs

# Generative-AI / LLM crawlers — explicitly allowed so the site can be cited

# with the same two private tools held back.
